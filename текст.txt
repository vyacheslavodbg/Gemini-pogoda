    // Haversine Distance Calculation (km)
    function calculateDistance(lat1, lon1, lat2, lon2) {
      const R = 6371; // Radius of the Earth in km
      const dLat = (lat2 - lat1) * Math.PI / 180;
      const dLon = (lon2 - lon1) * Math.PI / 180;
      const a = 
        Math.sin(dLat / 2) * Math.sin(dLat / 2) +
        Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * 
        Math.sin(dLon / 2) * Math.sin(dLon / 2);
      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
      return Math.round(R * c);
    }

    // Astronomical Moon Phase Calculation
    function getMoonPhaseData(date = new Date()) {
      let year = date.getFullYear();
      let month = date.getMonth() + 1;
      let day = date.getDate();

      if (month < 3) {
        year--;
        month += 12;
      }
      month++;

      let c = 365.25 * year;
      let e = 30.6 * month;
      let jd = c + e + day - 694039.09; // Julian Date relative to Jan 1900
      jd /= 29.5305882; // Divide by synodic month (29.53 days)
      let phase = jd - Math.floor(jd); // Fraction of current cycle
      let age = Math.round(phase * 29.53);
      let illumination = Math.round((1 - Math.cos(phase * 2 * Math.PI)) / 2 * 100);

      let phaseName = "";
      let icon = "🌔";

      if (age === 0 || age === 29) { phaseName = "Новолуние"; icon = "🌑"; }
      else if (age < 7) { phaseName = "Растущий серп"; icon = "🌒"; }
      else if (age === 7) { phaseName = "Первая четверть"; icon = "🌓"; }
      else if (age < 15) { phaseName = "Растущая Луна"; icon = "🌔"; }
      else if (age === 15) { phaseName = "Полнолуние"; icon = "🌕"; }
      else if (age < 22) { phaseName = "Убывающая Луна"; icon = "🌖"; }
      else if (age === 22) { phaseName = "Последняя четверть"; icon = "🌗"; }
      else { phaseName = "Убывающий серп"; icon = "🌘"; }

      // Calculate next full and new moon dates
      let daysUntilFull = (15 - age + 30) % 30;
      let daysUntilNew = (30 - age) % 30;

      let nextFull = new Date();
      nextFull.setDate(date.getDate() + daysUntilFull);

      let nextNew = new Date();
      nextNew.setDate(date.getDate() + daysUntilNew);

      return {
        phaseName,
        icon,
        age,
        illumination,
        nextFull: nextFull.toLocaleDateString('ru-RU', { day: 'numeric', month: 'long' }),
        nextNew: nextNew.toLocaleDateString('ru-RU', { day: 'numeric', month: 'long' })
      };
    }

    // Fetch Geomagnetic SWPC NOAA Data
    async function fetchGeomagneticData() {
      try {
        const url = `https://services.swpc.noaa.gov/products/noaa-planetary-k-index.json`;
        const res = await fetch(url);
        if (!res.ok) throw new Error("NOAA SWPC Error");
        const json = await res.json();
        
        // Skip header row if present
        const rows = Array.isArray(json[0]) && isNaN(parseFloat(json[0][1])) ? json.slice(1) : json;
        const recent = rows.slice(-10).map(row => {
          const rawTime = row[0];
          const dateObj = new Date(rawTime.replace(' ', 'T'));
          const timeStr = isNaN(dateObj.getTime()) 
            ? rawTime 
            : dateObj.toLocaleString('ru-RU', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' });
          const rawKp = parseFloat(row[1]);
          return {
            time: timeStr,
            kp: isNaN(rawKp) ? 2.0 : Math.round(rawKp * 10) / 10
          };
        });
        return recent.length > 0 ? recent : getGeomagneticFallback();
      } catch (err) {
        console.warn("Geomagnetic fetch fallback enabled:", err);
        return getGeomagneticFallback();
      }
    }

    function getGeomagneticFallback() {
      const now = new Date();
      const list = [];
      for (let i = 7; i >= 0; i--) {
        const d = new Date(now.getTime() - i * 3 * 3600 * 1000);
        const timeStr = d.toLocaleString('ru-RU', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' });
        const kpVal = Math.round((2.0 + Math.sin(i * 1.5) * 0.8) * 10) / 10;
        list.push({ time: timeStr, kp: Math.max(0, kpVal) });
      }
      return list;
    }
